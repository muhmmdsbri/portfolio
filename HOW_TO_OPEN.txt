🚀 كيفية فتح الموقع

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

الطريقة 1: النقر المزدوج (الأسهل)
─────────────────────────────────
1. ابحث عن ملف "index.html" في هذا المجلد
2. انقر عليه مرتين
3. سيفتح الموقع في متصفحك الافتراضي

الطريقة 2: من المتصفح
─────────────────────
1. افتح متصفحك (Chrome, Firefox, Safari, Edge)
2. اضغط Ctrl+O (أو Cmd+O على Mac)
3. اختر ملف "index.html"
4. انقر "فتح"

الطريقة 3: السحب والإفلات
──────────────────────────
1. افتح متصفحك
2. اسحب ملف "index.html" إلى نافذة المتصفح
3. سيفتح الموقع مباشرة

🆘 حل المشاكل الشائعة

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

المشكلة: الموقع لا يفتح
الحل: تأكد من وجود ملف index.html وحاول فتحه بمتصفح آخر

المشكلة: الصور لا تظهر
الحل: تأكد من وجود مجلد assets/images والملفات فيه

المشكلة: البيانات لا تحمل
الحل: تحقق من صحة JSON في data/portfolio-data.json

المشكلة: الموقع بطيء
الحل: ضغط الصور باستخدام TinyPNG
